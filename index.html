💡 الفكرة المبسطة

جسم الإنسان موصل كهربائي طبيعي. عندما يقترب من طرف GPIO مخصص للـ Touch، تتغير سعة (Capacitance) الطرف. ESP32 يقيس هذه السعة باستمرار، ويلاحظ الفرق فوراً.

📊 القيم: بدون لمس — القراءة عالية (مثل 80).
مع لمس — القراءة تنخفض (مثل 15).
لذلك: الشرط في الكود if (value < 30) يعني "تم اللمس".

🧪 الكود 1: اختبار اللمس

07 / 11

قبل استخدام الشاشة، نتأكد أن الحساس يعمل. سنقرأ قيم Touch على Serial Monitor.

// اختبار Touch Sensor — GPIO 4
#define TOUCH_PIN 4

void setup() {
  Serial.begin(115200);
  Serial.println("Touch Test Ready!");
}

void loop() {
  int value = touchRead(TOUCH_PIN);
  Serial.print("Touch value: ");
  Serial.println(value);

  if (value < 30) {
    Serial.println("  >> TOUCHED! 👆");
  }
  delay(200);
}
✅ بدون لمس: القيمة عادة 60-100
👆 عند اللمس: القيمة تنخفض إلى 10-25

📺 الكود 2: Touch + شاشة OLED

08 / 11 
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH  128
#define SCREEN_HEIGHT 64
#define TOUCH_PIN     4
#define TOUCH_LIMIT   30

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED failed!");
    while (1);
  }
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(2);
  display.setCursor(0, 20);
  display.println("Touch me!");
  display.display();
}

void loop() {
  int v = touchRead(TOUCH_PIN);

  display.clearDisplay();

  // العنوان
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("== ESP32 Touch ==");

  // القيمة
  display.setTextSize(2);
  display.setCursor(0, 15);
  display.print("Val: ");
  display.println(v);

  // الحالة
  display.setCursor(0, 45);
  if (v < TOUCH_LIMIT) {
    display.println("TOUCHED! 👆");
  } else {
    display.println("Waiting...");
  }

  display.display();
  delay(100);
}
🎯 3 أفكار مفتاحية:

Wire.begin(21, 22) — نحدد أطراف I2C يدوياً (SDA=21, SCL=22).
display.begin(..., 0x3C) — عنوان الشاشة على ناقل I2C.
display.display() — لا تنسها! بدونها الشاشة لا تتحدث.
